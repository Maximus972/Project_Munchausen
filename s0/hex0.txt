Прочитанный байт мы обрабатываем. Проверка что за символ.
lb t0, 0(a1)

Если # или ; то переходим в режим пропуска символов, до тех пор, пока не встретим символ \n. Когда встречаем символ \n, переключаемся обратно в старый режим.
addi t1, zero, 10
beq t0, t1 #;reset_flag

addi t1, zero, 59
beq t0, t1, #;set_flag

addi t1, zero, 35
beq  t0, t1, set_flag

; Это не \n и не ;, значит обрабатываем в зависимости от флага ;
addi t3, zero, 1
beq t2, t3, Чтение байта флаг ; равен 1 значит прыгаем в самое начало обратно

addi t1, zero, 48
blt t0, t1, Чтение байта

;отсекли ; со всей строкой, отсекли меньшее чем '0'
addi t1, zero, 58
blt t0, t1, digit_handler

; Проверка на 'A' - 'F'
addi t1, zero, 65
blt t0, t1, Чтение файла

addi t1, zero, 71
blt t0, t1, A-F handler

; Проверка на 'a' - 'f'

blt t0, t1, Чтение файла

addi t1, zero, 103
